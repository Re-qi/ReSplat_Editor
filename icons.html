原始代码示例

// 在 menu.ts 中使用
{
    text: localize('menu.help.github-repo'),
    icon: 'E259',  // 这是 Unicode 码点 0xE259
    onSelect: () => openUrl('https://github.com/Re-qi/ReSplat_Editor')
}

// JavaScript 中的转换代码（来自 menu-panel.ts）
